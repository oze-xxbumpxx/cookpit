### 1. .claude/settings.json — allow リスト + SessionStart hook 配線

`permissions` に `allow` を追加し、`hooks` に `SessionStart` を追加する
（`session-start-info.sh` は Phase 0 でコミット済み・未配線）:

```json
"permissions": {
  "allow": [
    "Bash(pnpm lint)", "Bash(pnpm lint:*)",
    "Bash(pnpm type-check)", "Bash(pnpm type-check:*)",
    "Bash(pnpm test)", "Bash(pnpm test:*)",
    "Bash(pnpm format)",
    "Bash(git status)", "Bash(git status:*)",
    "Bash(git diff:*)", "Bash(git log:*)",
    "Bash(git branch)", "Bash(git show:*)"
  ],
  "deny": [ ...現状のまま... ]
},
"hooks": {
  "SessionStart": [
    { "hooks": [ { "type": "command",
        "command": "bash \"$CLAUDE_PROJECT_DIR/.claude/hooks/session-start-info.sh\"" } ] }
  ],
  ...既存の PreToolUse 以下はそのまま...
}
```

### 2. guard-dangerous.mjs — 保護構成ファイルの書き込みを警告→ブロックに格上げ

`import { readFileSync, existsSync } from 'node:fs'; import { join } from 'node:path';` とし、
`const ROOT = process.env.CLAUDE_PROJECT_DIR || process.cwd();` を追加のうえ、以下を挿入：

```js
// ── 保護構成ファイルへの書き込み判定（improvement-cycle.md §承認境界）─────
// 承認マーカーが無い限り構成ファイルへの Edit/Write/MultiEdit をブロックする。
// 「read-only を宣言する Agent が Write を保持する」穴を決定論的に塞ぐ（2026-07-04 監査）。
// .claude/evals/ と .claude/state/ は成果物置き場のため対象外。
function isProtectedConfigPath(p) {
  if (typeof p !== 'string') return false;
  const rel = p.startsWith(ROOT) ? p.slice(ROOT.length + 1) : p;
  return (
    rel === 'CLAUDE.md' ||
    rel === '.claude/settings.json' ||
    rel.startsWith('.claude/agents/') ||
    rel.startsWith('.claude/hooks/') ||
    rel.startsWith('.claude/rules/') ||
    rel.startsWith('.claude/skills/')
  );
}
function configChangeApproved() {
  return existsSync(join(ROOT, '.claude/state/config-change-approved'));
}
```

`main()` の Edit/Write/MultiEdit 分岐に追加：

```js
if (tool !== 'Read' && isProtectedConfigPath(ti.file_path) && !configChangeApproved()) {
  deny(
    `保護構成ファイルへの未承認の書き込み: ${ti.file_path}\n` +
      `人間が承認する場合は .claude/state/config-change-approved を作成してから再実行\n` +
      `（touch .claude/state/config-change-approved。作業後は削除する）`,
  );
}
```

適用後の運用：構成変更を承認するときは `touch .claude/state/config-change-approved`、
完了後に削除。validate-agent-config.mjs（警告）と二層になる。
